import React, { useEffect, useState } from 'react';

export const PageGrid = () => {
  const [cards, setCards] = useState([]);
  const [others, setOthers] = useState([]);
  const [error, setError] = useState(null);
  const [loaderStarted, setLoaderStarted] = useState(false);

  const iconMap = [
    { keywords: ['history', 'map', 'archive'], icon: '📜' },
    { keywords: ['article', 'story', 'news', 'blog'], icon: '📰' },
    { keywords: ['guide', 'manual', 'book'],          icon: '📖' },
    { keywords: [],                                    icon: '📄' },
  ];
  const pickIcon = (title) => {
    const t = title.toLowerCase();
    for (let { keywords, icon } of iconMap) {
      if (keywords.some(k => t.includes(k))) return icon;
    }
    return iconMap[iconMap.length - 1].icon;
  };

  // kick off our 5s bar fill
  useEffect(() => {
    setLoaderStarted(true);
  }, []);

  // your existing fetch logic
  useEffect(() => {
    if (typeof document === 'undefined') return;
    (async () => {
      try {
        const nav = document.querySelector('.rm-Sidebar');
        if (!nav) throw new Error('Sidebar not found');
        const active = nav.querySelector(
          'a.rm-Sidebar-link[aria-current="page"], a.rm-Sidebar-link.active'
        );
        if (!active) throw new Error('No active page link detected');
        const currentHref = active.href;
        const section = active.closest('section.rm-Sidebar-section');
        if (!section) throw new Error('Unable to locate category section');
        const anchors = Array.from(
          section.querySelectorAll('a.rm-Sidebar-link[href^="/docs/"]')
        );
        const raw = anchors.map(a => ({
          href: a.href,
          title: a.textContent.trim(),
        }));
        const seen = new Set();
        const pages = raw
          .filter(item => {
            if (seen.has(item.href)) return false;
            seen.add(item.href);
            return true;
          })
          .filter(item => item.href !== currentHref);
        const toc = document.querySelector('.content-toc');
        if (toc) toc.style.display = 'none';

        const slice = pages.slice(0, 8);
        const enriched = await Promise.all(
          slice.map(item =>
            fetch(item.href)
              .then(r => r.text())
              .then(html => {
                const doc = new DOMParser().parseFromString(html, 'text/html');
                const img = doc.querySelector('article img, .rm-Page-body img');
                const p   = doc.querySelector('article p, .rm-Page-body p');
                return {
                  ...item,
                  image: img ? img.src : null,
                  excerpt: p ? p.textContent.trim().slice(0, 120) : '',
                };
              })
              .catch(() => ({ ...item, image: null, excerpt: '' }))
          )
        );

        setCards(enriched);
        setOthers(pages.slice(8));
      } catch (e) {
        setError(e.message);
      }
    })();
  }, []);

  if (error) {
    return <p style={{ color: 'red' }}>Error loading pages: {error}</p>;
  }

  // **Pure 5s loader** — no progress state
  if (!cards.length && !others.length) {
    return (
      <div className="retro-loader">
        <div className="bar">
          <div
            className="fill"
            style={{
              width: loaderStarted ? '100%' : '0',
              transition: 'width 5s linear'
            }}
          />
        </div>
        <div className="message">🤖 Retrieving Family Information...</div>
        <style>{`
          .retro-loader {
            font-family: 'Courier New', Courier, monospace;
            text-align: center;
            margin: 60px 0;
          }
          .retro-loader .bar {
            width: 280px;
            height: 16px;
            border: 3px solid #222;
            margin: 0 auto 12px;
            background: #fff;
            box-shadow: inset 0 0 4px rgba(0,0,0,0.3);
            overflow: hidden;
          }
          .retro-loader .fill {
            height: 100%;
            background: #222;
          }
          .retro-loader .message {
            font-size: 0.95rem;
            color: #222;
          }
        `}</style>
      </div>
    );
  }

  // your existing grid + link-list
  return (
    <>
      <style>{`
        .page-grid {
          display: grid;
          grid-template-columns: repeat(auto-fit, minmax(220px,1fr));
          gap:20px;
          margin:24px 0;
        }
        /* … all your existing styles … */
      `}</style>
      <div className="page-grid">
        {cards.map((l, i) => (
          <div key={i} className="page-tile">
            {l.image
              ? <img src={l.image} alt={`Preview for ${l.title}`} />
              : <div className="placeholder"><i className="fa fa-image" /></div>
            }
            <div className="page-header">
              <span className="page-icon">{pickIcon(l.title)}</span>
              <h4 className="page-title">{l.title}</h4>
            </div>
            {l.excerpt && <p>{l.excerpt}…</p>}
            <a href={l.href}>View</a>
          </div>
        ))}
      </div>
      {others.length > 0 && (
        <>
          <div className="more-heading">More…</div>
          <div className="link-list">
            {others.map((o, idx) => (
              <a key={idx} href={o.href}>{o.title}</a>
            ))}
          </div>
        </>
      )}
    </>
  );
};

<PageGrid />
