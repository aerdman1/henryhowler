import React, { useEffect, useState } from "react";

export const ChangelogEmbed = ({ src }) => {
  const [html, setHtml] = useState("");
  const [error, setError] = useState("");

  useEffect(() => {
    let isMounted = true;
    const controller = new AbortController();

    async function loadChangelog() {
      try {
        setError("");
        setHtml("");

        // IMPORTANT: omit credentials so we fetch the public (logged-out) view
        const res = await fetch(src, {
          method: "GET",
          credentials: "omit",
          cache: "no-store",
          redirect: "follow",
          signal: controller.signal,
          headers: {
            Accept: "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
          },
        });

        if (!res.ok) throw new Error(`Failed to fetch: ${res.status}`);
        const text = await res.text();

        // Parse HTML and extract only the changelog <main> content
        const parser = new DOMParser();
        const doc = parser.parseFromString(text, "text/html");

        const main =
          doc.querySelector("main.rm-Changelog") ||
          doc.querySelector("main#content.rm-Changelog") ||
          doc.querySelector("main#content");

        const articles = main ? main.innerHTML : "";

        if (!articles) throw new Error("No changelog content found on the page.");

        if (isMounted) setHtml(articles);
      } catch (err) {
        if (err && err.name === "AbortError") return;
        console.error("Failed to load changelog:", err);
        if (isMounted) setError("Could not load changelog content.");
      }
    }

    loadChangelog();

    return () => {
      isMounted = false;
      controller.abort();
    };
  }, [src]);

  if (error) {
    return <p className="text-red-500">{error}</p>;
  }

  if (!html) {
    return <p className="opacity-70">Loading changelog…</p>;
  }

  return (
    <div
      className="prose dark:prose-invert max-w-none border border-gray-300 dark:border-gray-700 rounded-md p-4 bg-white dark:bg-gray-800"
      dangerouslySetInnerHTML={{ __html: html }}
    />
  );
};

<ChangelogEmbed src="https://family.henryhowler.com/changelog" />
