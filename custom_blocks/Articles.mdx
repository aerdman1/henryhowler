import React, { useEffect, useState } from "react";

export const ChangelogEmbed = ({
  src,
  maxPosts = 3,
  maxCharsPerPost = 350,
  showReadMore = true,
}) => {
  const [html, setHtml] = useState("");
  const [error, setError] = useState("");

  useEffect(() => {
    let isMounted = true;
    const controller = new AbortController();

    function safeText(v) {
      if (v === null || v === undefined) return "";
      return String(v);
    }

    function truncateText(text, maxChars) {
      const t = safeText(text).replace(/\s+/g, " ").trim();
      if (!maxChars || maxChars <= 0) return t;
      if (t.length <= maxChars) return t;
      return t.slice(0, maxChars).replace(/\s+\S*$/, "").trim() + "…";
    }

    async function loadChangelog() {
      try {
        setError("");
        setHtml("");

        const res = await fetch(src, {
          method: "GET",
          credentials: "omit", // public view even when you're logged in elsewhere
          cache: "no-store",
          redirect: "follow",
          signal: controller.signal,
          headers: {
            Accept: "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
          },
        });

        if (!res.ok) throw new Error(`Failed to fetch: ${res.status}`);
        const text = await res.text();

        const parser = new DOMParser();
        const doc = parser.parseFromString(text, "text/html");

        const main =
          doc.querySelector("main.rm-Changelog") ||
          doc.querySelector("main#content.rm-Changelog") ||
          doc.querySelector("main#content") ||
          doc.querySelector("main");

        if (!main) throw new Error("No <main> found on changelog page.");

        // Grab posts (ReadMe changelog uses <article> for each post)
        const articleNodes = Array.from(main.querySelectorAll("article"));
        if (!articleNodes.length) throw new Error("No posts found on changelog page.");

        const limited = articleNodes.slice(0, Math.max(1, Number(maxPosts) || 3));

        // Rewrite each post body to a safe excerpt + optional read-more link
        for (let i = 0; i < limited.length; i++) {
          const article = limited[i];

          const body =
            article.querySelector(".rm-Markdown") ||
            article.querySelector(".markdown-body") ||
            article.querySelector("section") ||
            article;

          // Find canonical link for "Read more"
          const a =
            article.querySelector('h1 a[href], h2 a[href], a[href*="/changelog/"]') ||
            article.querySelector("a[href]");
          let href = a ? a.getAttribute("href") : "";
          if (href && href.startsWith("/")) {
            try {
              href = new URL(href, src).toString();
            } catch (e) {}
          }

          // Keep images if present, but clamp text
          const firstImgWrap = body.querySelector(".img.lightbox") || body.querySelector("img")?.closest(".img.lightbox");
          const imgHtml = firstImgWrap ? firstImgWrap.outerHTML : "";

          // Build excerpt text from paragraphs (fallback to body textContent)
          const pNodes = Array.from(body.querySelectorAll("p"));
          const combinedText = pNodes.length
            ? pNodes.map(p => safeText(p.textContent)).join("\n\n")
            : safeText(body.textContent);

          const excerpt = truncateText(combinedText, Number(maxCharsPerPost) || 350);

          // Replace body content with excerpt (prevents huge blocks)
          if (body) {
            body.innerHTML =
              imgHtml +
              `<p class="hh-changelog-excerpt">${excerpt}</p>` +
              (showReadMore && href
                ? `<p class="hh-changelog-more"><a href="${href}" target="_blank" rel="noopener noreferrer">Read more →</a></p>`
                : "");
          }
        }

        // Render only the limited posts (not the whole main)
        const wrapper = doc.createElement("div");
        wrapper.className = "hh-changelog-list";
        limited.forEach(a => wrapper.appendChild(a.cloneNode(true)));

        if (isMounted) setHtml(wrapper.innerHTML);
      } catch (err) {
        if (err && err.name === "AbortError") return;
        console.error("Failed to load changelog:", err);
        if (isMounted) setError("Could not load changelog content.");
      }
    }

    loadChangelog();

    return () => {
      isMounted = false;
      controller.abort();
    };
  }, [src, maxPosts, maxCharsPerPost, showReadMore]);

  if (error) return <p className="text-red-500">{error}</p>;
  if (!html) return <p className="opacity-70">Loading changelog…</p>;

  return (
    <div
      className="prose dark:prose-invert max-w-none border border-gray-300 dark:border-gray-700 rounded-md p-4 bg-white dark:bg-gray-800"
      dangerouslySetInnerHTML={{ __html: html }}
    />
  );
};


<ChangelogEmbed
  src="https://family.henryhowler.com/changelog"
  maxPosts={3}
  maxCharsPerPost={320}
  showReadMore={true}
/>;
